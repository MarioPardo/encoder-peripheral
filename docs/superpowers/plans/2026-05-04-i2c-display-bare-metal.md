### Task 2: Create ssd1306.h

**Files:**
- Create: `vitis_workspace/phase1_test/src/ssd1306.h`

- [ ] **Step 1: Write driver header**

```c
#ifndef SSD1306_H
#define SSD1306_H

#include <string.h>
#include "xiicps.h"
#include "ssd1306_font.h"

#define SSD1306_ADDR   0x3C
#define SSD1306_I2C_HZ 10000

/* Send a single command byte */
static inline int ssd1306_cmd(XIicPs *iic, uint8_t cmd)
{
    uint8_t buf[2] = {0x00, cmd};
    return XIicPs_MasterSendPolled(iic, buf, 2, SSD1306_ADDR);
}

/* Send a block of pixel data for one page */
static inline int ssd1306_data(XIicPs *iic, uint8_t *data, int len)
{
    /* XIicPs_MasterSendPolled needs a mutable buffer with control byte prepended */
    uint8_t buf[129];
    buf[0] = 0x40;
    memcpy(&buf[1], data, len);
    return XIicPs_MasterSendPolled(iic, buf, len + 1, SSD1306_ADDR);
}

static inline void ssd1306_init(XIicPs *iic)
{
    XIicPs_SetSClk(iic, SSD1306_I2C_HZ);

    ssd1306_cmd(iic, 0xAE); /* display off */
    ssd1306_cmd(iic, 0x20); ssd1306_cmd(iic, 0x00); /* horizontal addressing */
    ssd1306_cmd(iic, 0xB0); /* page start 0 */
    ssd1306_cmd(iic, 0xC8); /* COM scan direction */
    ssd1306_cmd(iic, 0x00); /* column low */
    ssd1306_cmd(iic, 0x10); /* column high */
    ssd1306_cmd(iic, 0x40); /* start line */
    ssd1306_cmd(iic, 0x81); ssd1306_cmd(iic, 0xFF); /* contrast max */
    ssd1306_cmd(iic, 0xA1); /* segment remap */
    ssd1306_cmd(iic, 0xA6); /* normal display */
    ssd1306_cmd(iic, 0xA8); ssd1306_cmd(iic, 0x3F); /* mux ratio */
    ssd1306_cmd(iic, 0xA4); /* output from RAM */
    ssd1306_cmd(iic, 0xD3); ssd1306_cmd(iic, 0x00); /* display offset */
    ssd1306_cmd(iic, 0xD5); ssd1306_cmd(iic, 0xF0); /* clock div */
    ssd1306_cmd(iic, 0xD9); ssd1306_cmd(iic, 0x22); /* pre-charge */
    ssd1306_cmd(iic, 0xDA); ssd1306_cmd(iic, 0x12); /* COM pins */
    ssd1306_cmd(iic, 0xDB); ssd1306_cmd(iic, 0x20); /* VCOMH */
    ssd1306_cmd(iic, 0x8D); ssd1306_cmd(iic, 0x14); /* charge pump on */
    ssd1306_cmd(iic, 0xAF); /* display on */
}

static inline void ssd1306_clear(XIicPs *iic)
{
    uint8_t blank[128];
    memset(blank, 0, sizeof(blank));
    for (int page = 0; page < 8; page++) {
        ssd1306_cmd(iic, 0xB0 | page);
        ssd1306_cmd(iic, 0x00); /* col low */
        ssd1306_cmd(iic, 0x10); /* col high */
        ssd1306_data(iic, blank, 128);
    }
}

/* Draw string at given page (0-7) and starting column (0-127) */
static inline void ssd1306_draw_string(XIicPs *iic, uint8_t col, uint8_t page,
                                        const char *str)
{
    ssd1306_cmd(iic, 0xB0 | page);
    ssd1306_cmd(iic, col & 0x0F);        /* col addr low */
    ssd1306_cmd(iic, 0x10 | (col >> 4)); /* col addr high */

    while (*str) {
        uint8_t c = (uint8_t)*str++;
        if (c < 0x20 || c > 0x7E) c = 0x20;
        uint8_t glyph[6];
        memcpy(glyph, font5x7[c - 0x20], 5);
        glyph[5] = 0x00; /* column spacer */
        ssd1306_data(iic, glyph, 6);
    }
}

#endif /* SSD1306_H */
```

```c
#include <stdio.h>
#include <stdint.h>
#include "xil_printf.h"
#include "xparameters.h"
#include "xiicps.h"
#include "ssd1306.h"

#define ENC_BASE     XPAR_ENCODER_AXI_0_BASEADDR

#define REG_CTRL     *((volatile uint32_t *)(ENC_BASE + 0x00))
#define REG_STATUS   *((volatile uint32_t *)(ENC_BASE + 0x04))
#define REG_POSITION *((volatile int32_t  *)(ENC_BASE + 0x08))
#define REG_VELOCITY *((volatile int32_t  *)(ENC_BASE + 0x0C))

#define CTRL_ENABLE   (1 << 0)
#define CTRL_CLR_POS  (1 << 1)

int main(void)
{
    xil_printf("\r\n=== Step 1: I2C Display Test ===\r\n");

    /* Init PS I2C */
    XIicPs iic;
    XIicPs_Config *cfg = XIicPs_LookupConfig(XPAR_XIICPS_0_DEVICE_ID);
    XIicPs_CfgInitialize(&iic, cfg, cfg->BaseAddress);
    XIicPs_SetSClk(&iic, SSD1306_I2C_HZ);

    xil_printf("I2C init done. Initializing display...\r\n");
    ssd1306_init(&iic);
    ssd1306_clear(&iic);
    ssd1306_draw_string(&iic, 0, 0, "HELLO ZYNQ");
    xil_printf("Display init done. Check OLED.\r\n");

    while (1);

    return 0;
}
```
